* Built-in .sort() method

** Example

#+BEGIN_SRC java
import java.util.*; // import whole package including Arrays
import java.util.Arrays; // import just Arrays

int[] numbers = { 6, 9, 1 };
Arrays.sort(numbers);

for (int i = 0; i < numbers.length; i++)
    System.out.print(numbers[i] + " ");

String[] names = { "DeadPool", "BatMan", "AquaMan" };

Arrays.sort(names);

for (String name: names)
    System.out.println(name);

#+END_SRC

** Example (Descending)

#+BEGIN_SRC java
import java.util.Arrays;
import java.util.Collections;

public class DescendingSortExample {

    public static void main(String[] args) {
        Integer[] numbers = {5, 3, 8, 6, 7, 2};

        // Sorting in descending order (note: Integer[] is used for this purpose)
        Arrays.sort(numbers, Collections.reverseOrder());

        System.out.println("Sorted Array in Descending Order: " + Arrays.toString(numbers));
    }
}
#+END_SRC
